═══════════════════════════════════════════════════════════════════════
                        PLAYERSTATS OBJECT
═══════════════════════════════════════════════════════════════════════

PlayerStats {
  // Raw Hitting Stats
  atBats: number
  runs: number
  hits: number
  doubles: number
  triples: number
  homeRuns: number
  walks: number
  hitByPitch: number
  sacrificeHits: number
  stolenBases: number
  runsBattedIn: number
  strikeouts: number
  
  // Pitching Stats
  inningsPitched: number
  earnedRuns: number
  totalPitches: number
  
  // Advanced Stats
  twoStrikeCounts: number
  successWithTwoStrikes: number
  
  // Calculated Stats (computed, not stored)
  battingAverage: number          // hits / atBats
  onBasePercentage: number        // (hits + walks + HBP) / (atBats + walks + HBP + SH)
  sluggingPercentage: number      // (singles + 2×doubles + 3×triples + 4×HR) / atBats
  onBasePlusSlugging: number      // OBP + SLG
  isolatedPower: number           // SLG - AVG
  runsCreated: number             // ((hits + walks) × totalBases) / (atBats + walks)
  earnedRunAverage: number        // (earnedRuns × 9) / inningsPitched
  strikeoutRate: number           // strikeouts / atBats
  walksHitsPerInningPitched: number  // (walks + hits) / inningsPitched
  twoStrikeSuccessRate: number    // successWithTwoStrikes / twoStrikeCounts
  war: number                     // Wins Above Replacement
}

═══════════════════════════════════════════════════════════════════════
                       QUERY PATTERNS
═══════════════════════════════════════════════════════════════════════

1. Get Active Season:
   users/{userId}/seasons/ WHERE isActive == true

2. Get Season Games (chronological):
   users/{userId}/seasons/{seasonId}/games/ ORDER BY date ASC

3. Get Cached AI Insight:
   users/{userId}/stats/{statName}/insights/ (latest document)

4. Get Team Games:
   games/ WHERE teamHome == {teamId} OR teamAway == {teamId}

5. Aggregate Season Stats:
   - Read all games in season
   - Sum raw stats (hits, atBats, etc.)
   - Calculate cumulative stats (batting average, etc.)
